# 0. BOOTSTRAP (one line, then you have OTS_BIN, PY, REPO in your shell)

eval "$(bash tools/bootstrap.sh --export)"

# 1. AUDIT (read-only; safe to run anytime; ~3 seconds)

python3 scripts/selfcheck_master.py
# expected: "OK: 7/7 checks pass on 4774 files"

# 2. STAMP (anchors current bytes in OTS calendars; ~5-15 seconds)

python3 scripts/stamp_master_package.py
# writes MASTER/MASTER_PACKAGE.ots (real PendingAttestation, 4 calendars)
# writes MASTER/MASTER_PACKAGE.json (ots_status = STAMPED_PENDING_BITCOIN)

# 3. REINDEX (captures post-stamp hashes; <1 second)

python3 scripts/rebuild_master_files.py
# writes MASTER/MASTER_FILES.json, INVENTORY.csv, MANIFEST.sha256

# 4. SELF-CHECK (the 7 assertions; <1 second)

python3 scripts/selfcheck_master.py
# MUST still say "7/7 checks pass". If it fails on step 6, you ran
# rebuild before stamp. Re-run stamp then rebuild (order matters).

# 5. COMMIT (only after 7/7)

git add MASTER/ scripts/ tools/
git commit -m "what changed — reindexed, re-stamped, verified"

# 6. UPGRADE OTS TO BITCOIN (slow, optional)

python3 scripts/upgrade_ots.py \
  --root-file MASTER/02_unsigned_pile/unsigned_merkle_root.json \
  --stamp-file MASTER/02_unsigned_pile/unsigned_merkle_root.json.ots \
  --wait
# --wait blocks until a Bitcoin block header attestation is available
# without --wait, the script exits idempotently with the current state

# 7. ADD A FILE TO THE MASTER (operator workflow)

cp /path/to/new.md MASTER/07_crosswalk/
python3 scripts/stamp_master_package.py    # stamp FIRST
python3 scripts/rebuild_master_files.py   # rebuild SECOND
python3 scripts/selfcheck_master.py        # assert THIRD
# expected: "OK: 7/7"
git add MASTER/ scripts/
git commit -m "add <description>"
